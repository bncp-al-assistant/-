/** * CLUB HOUSE & LOUNGE 예약 신청 Worker * ----------------------------------------------------- * KV Binding 이름: CLUBHOUSE_KV (wrangler.toml 에서 반드시 이 이름으로 연결) * * 제공 API * GET /api/reservations -> 전체 예약 목록 조회 * GET /api/reservations?date=YYYY-MM-DD -> 특정 날짜 예약 조회 * POST /api/reservations -> 예약 신청 저장 * GET /api/meta -> 팀/장소/기본품목 등 폼 메타데이터 */ // ---- 프론트엔드(폼)에서 쓰는 고정 메타데이터 ---------------------------- const META = { teams: ["공사팀", "사업관리팀", "관리재경팀", "경호경비팀", "이라크 지사", "기타"], locations: [ "Club House Hall", "Club House 1-1", "Club House 1-2", "Club House 1-3", "Club House 1-4", "Club House 1-5", "Club House 1-6", "Lounge HB-01", "Lounge HM-20", "Lounge HM-28", "Lounge HM-29", "Lounge HM-30", "Lounge HM-31", ], defaultProvidedItems: ["야채", "쌈장", "기름장", "석식메뉴"], }; // GitHub Pages 등 외부 오리진에서 호출할 수 있도록 CORS 허용 function corsHeaders(origin) { return { "Access-Control-Allow-Origin": origin || "*", "Access-Control-Allow-Methods": "GET, POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", "Access-Control-Max-Age": "86400", }; } function json(data, status = 200, origin) { return new Response(JSON.stringify(data), { status, headers: { "Content-Type": "application/json; charset=utf-8", ...corsHeaders(origin), }, }); } function badRequest(message, origin) { return json({ ok: false, error: message }, 400, origin); } // 간단한 유효성 검사 (reservationFormSchema 의 required 항목 기준) function validateReservation(body) { const errors = []; if (!body.team) errors.push("팀명(team)이 필요합니다."); if (!body.applicant) errors.push("신청자(applicant)가 필요합니다."); if (!body.usageDate) errors.push("이용일자(usageDate)가 필요합니다."); if (!body.headcount || Number(body.headcount) < 1) errors.push("사용 인원수(headcount)는 1 이상이어야 합니다."); if (!body.location) errors.push("장소(location)가 필요합니다."); return errors; } export default { async fetch(request, env, ctx) { const url = new URL(request.url); const origin = request.headers.get("Origin"); // Preflight if (request.method === "OPTIONS") { return new Response(null, { headers: corsHeaders(origin) }); } // ---- KV 바인딩 존재 여부를 가장 먼저 확인 (여기서 이전 오류가 발생했음) ---- if (!env.CLUBHOUSE_KV) { return json( { ok: false, error: "CLUBHOUSE_KV 바인딩이 설정되지 않았습니다. Cloudflare 대시보드 > Workers & Pages > 해당 Worker > Settings > Bindings 에서 KV Namespace를 'CLUBHOUSE_KV' 라는 이름으로 연결해주세요.", }, 500, origin ); } try { // ---- 메타데이터 ---- if (url.pathname === "/api/meta" && request.method === "GET") { return json({ ok: true, data: META }, 200, origin); } // ---- 예약 목록 조회 ---- if (url.pathname === "/api/reservations" && request.method === "GET") { const dateFilter = url.searchParams.get("date"); const list = await env.CLUBHOUSE_KV.list({ prefix: "reservation:" }); const items = await Promise.all( list.keys.map(async (k) => { const raw = await env.CLUBHOUSE_KV.get(k.name); return raw ? JSON.parse(raw) : null; }) ); let reservations = items.filter(Boolean); if (dateFilter) { reservations = reservations.filter((r) => r.usageDate === dateFilter); } // 최신순 정렬 reservations.sort((a, b) => (b.createdAt || "").localeCompare(a.createdAt || "")); return json({ ok: true, count: reservations.length, data: reservations }, 200, origin); } // ---- 예약 신청 저장 ---- if (url.pathname === "/api/reservations" && request.method === "POST") { let body; try { body = await request.json(); } catch (e) { return badRequest("요청 본문이 올바른 JSON이 아닙니다.", origin); } const errors = validateReservation(body); if (errors.length > 0) { return json({ ok: false, error: errors.join(" ") }, 400, origin); } const id = crypto.randomUUID(); const createdAt = new Date().toISOString(); const reservation = { id, team: body.team, applicant: body.applicant, usageDate: body.usageDate, headcount: Number(body.headcount), location: body.location, providedItems: Array.isArray(body.providedItems) ? body.providedItems : META.defaultProvidedItems, additionalRequest: body.additionalRequest || "", remarks: body.remarks || "", createdAt, }; await env.CLUBHOUSE_KV.put(`reservation:${createdAt}:${id}`, JSON.stringify(reservation)); return json({ ok: true, data: reservation }, 201, origin); } return json({ ok: false, error: "지원하지 않는 경로입니다." }, 404, origin); } catch (err) { return json({ ok: false, error: `서버 오류: ${err.message}` }, 500, origin); } }, };
